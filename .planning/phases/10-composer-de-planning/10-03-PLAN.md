                {

                    "sprint_numero": sprint_numero,
                    "step_atual": 1,
                    "dados_json": {},
                },
                on_conflict="project_id,sprint_numero",

        fetch_resp = (

            .execute()
        )
        if not fetch_resp.data:
            raise HTTPException(status_code=500, detail="Falha ao criar/buscar rascunho")
        rascunho = fetch_resp.data[0]
