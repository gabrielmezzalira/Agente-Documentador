    export interface BoletimResponse {
      id: string;
      project_id: string;
      sprint_numero?: number | null;
      funcionalidade_ids: string[];
      status: "rascunho" | "enviado" | "aprovado" | "ajuste";

      conteudo: string;
      criado_em: string;
      enviado_em?: string | null;
      retorno_em?: string | null;
    }

Função `listBoletins(projectId: string): Promise<BoletimResponse[]>`:
- GET `${BASE}/boletins/${projectId}`
- if !res.ok throw new Error(await res.text())
- return res.json()

Função `createBoletim(body: { project_id: string; sprint_numero?: number | null; funcionalidade_ids: string[] }): Promise<BoletimResponse>`:
- POST `${BASE}/boletins` com JSON body
- if !res.ok throw new Error(await res.text())
- return res.json()

Função `patchBoletim(id: string, body: { status: string; retorno_tipo?: string }): Promise<BoletimResponse>`:
- PATCH `${BASE}/boletins/${id}` com JSON body
- if !res.ok throw new Error(await res.text())
- return res.json()

Função `gerarResumoSemanal(projectId: string): Promise<{ content: string }>`:
- POST `${BASE}/boletins/resumo_semanal` com JSON body `{ project_id: projectId }`
- if !res.ok throw new Error(await res.text())
- return res.json()

Seguir exatamente o padrão de indentação e error handling das funções existentes no arquivo.
