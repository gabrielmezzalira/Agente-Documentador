       @router.post("/importar", response_model=ImportPropostaResponse)
       async def importar_funcionalidades(data: ImportarRequest):
           """Propõe funcionalidades a partir do texto do contrato via IA. Não salva nada."""
           # Input guard: truncate at 50000 chars (DoS mitigation per T-07-03-03)
           texto = data.texto_contrato[:50000]
           if len(texto.strip()) < 20:
               raise HTTPException(status_code=422, detail="Texto do contrato muito curto para análise")

           # Fetch gemini_api_key for the project (with env fallback per PATTERNS.md)
           client = get_client()
           proj = client.table("projects").select("gemini_api_key").eq("id", data.project_id).execute()
           if not proj.data:
               raise HTTPException(status_code=404, detail="Project not found")
           api_key = (proj.data[0].get("gemini_api_key") or "").strip()
           if not api_key:
               api_key = os.environ.get("GEMINI_API_KEY", "")
           if not api_key:
               raise HTTPException(status_code=500, detail="Nenhuma API key Gemini configurada para este projeto")

           # Invoke the graph — async invoke
           state = await import_graph.ainvoke({
               "texto_contrato": texto,
               "projeto_id": data.project_id,
               "gemini_api_key": api_key,
               "proposta": None,
               "valido": False,
               "tentativas": 0,
               "erro": None,
           })

           if not state.get("valido") or not state.get("proposta"):
               raise HTTPException(
                   status_code=502,
                   detail=f"Não foi possível extrair funcionalidades do contrato: {state.get('erro', 'resposta inválida')}",
               )

           # Convert raw dicts to FuncionalidadeProposta (validates prioridade default etc.)
           propostas = []
           for item in state["proposta"]:
               try:
                   propostas.append(FuncionalidadeProposta(**item))
               except Exception:
                   # Skip malformed items silently — partial results are acceptable
                   pass

           if not propostas:
               raise HTTPException(status_code=502, detail="Gemini retornou lista sem itens válidos")

           return ImportPropostaResponse(propostas=propostas)

       @router.post("/importar/confirmar", response_model=list[FuncionalidadeResponse], status_code=201)
       async def confirmar_importacao(data: ImportConfirmarRequest):
           """Cria as funcionalidades confirmadas e descarta as rejeitadas."""
           client = get_client()
           # Verify project exists
           proj_check = client.table("projects").select("id").eq("id", data.project_id).execute()
           if not proj_check.data:
               raise HTTPException(status_code=404, detail="Project not found")

           confirmadas = [item.proposta for item in data.itens if item.confirmed]
           if not confirmadas:
               return []  # All rejected — valid scenario, return empty list

           criadas = []
           for prop in confirmadas:
               payload = {
                   "project_id": data.project_id,
                   "id_funcional": prop.id_funcional,
                   "titulo": prop.titulo,
                   "criterios_aceite": prop.criterios_aceite,
                   "prioridade": prop.prioridade,
               }
               if prop.descricao:
                   payload["descricao"] = prop.descricao
               resp = client.table("funcionalidades").insert(payload).execute()
               if resp.data:
                   criadas.append(resp.data[0])

           return criadas

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client → POST /funcionalidades/importar | texto_contrato is user-supplied free text forwarded to Gemini; project_id is untrusted UUID |
| FastAPI → Gemini API | texto_contrato crosses to external LLM — prompt injection possible |
| Gemini response → supabase INSERT | Gemini-generated content (id_funcional, titulo, criterios_aceite) is used in INSERT payloads |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-07-03-01 | Tampering | POST /funcionalidades/importar — prompt injection via texto_contrato | medium | mitigate | Gemini receives texto_contrato inside a HumanMessage after a SystemMessage that constrains the output format to JSON. The structured JSON output requirement limits the injection surface. The confirmar step requires human review before any data lands in DB — no injection reaches the DB without human approval. |
| T-07-03-02 | Information Disclosure | gemini_api_key fetched from DB and passed to import_graph state | high | mitigate | api_key is injected into ImportState (in-memory dict) for the duration of the request only. It is NOT returned in any API response. The graph's proposta output does not include the api_key field. Verify: grep the endpoint response models — ImportPropostaResponse and list[FuncionalidadeResponse] have no gemini_api_key field. |
| T-07-03-03 | Denial of Service | POST /funcionalidades/importar — very large texto_contrato sent to Gemini | medium | mitigate | Handler truncates texto_contrato to 50000 chars before invoking the graph (per action instructions). Gemini context window cost is bounded. |
| T-07-03-04 | Tampering | POST /importar/confirmar — inserting Gemini-generated id_funcional values | low | accept | supabase-py uses parameterized inserts — no SQL injection. Duplicate id_funcional values in the same project are allowed in MVP (no unique constraint yet — per RESEARCH.md Open Question 2). |
| T-07-03-05 | Repudiation | No record of which proposals were rejected | low | accept | Only confirmed items are saved. Rejected proposals leave no trace — acceptable for MVP scope. Audit of rejections is a post-MVP concern. |
</threat_model>
